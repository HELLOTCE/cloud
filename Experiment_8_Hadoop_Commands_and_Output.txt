
# EXPERIMENT 8: HADOOP AND MAPREDUCE
# COMMANDS AND OUTPUT

# STEP 1: INSTALL JAVA AND SSH
sudo apt update
sudo apt install openjdk-11-jdk openssh-server wget -y
java -version

# STEP 2: EXTRACT HADOOP (ALREADY DOWNLOADED)
cd ~/Downloads
tar -xzf hadoop-3.3.6.tar.gz
sudo mv hadoop-3.3.6 /usr/local/hadoop
sudo chown -R "$USER":"$USER" /usr/local/hadoop

# STEP 3: CONFIGURE ENVIRONMENT VARIABLES
echo 'export JAVA_HOME=/usr/lib/jvm/java-11-openjdk-amd64' >> ~/.bashrc
echo 'export HADOOP_HOME=/usr/local/hadoop' >> ~/.bashrc
echo 'export PATH=$PATH:$HADOOP_HOME/bin:$HADOOP_HOME/sbin' >> ~/.bashrc
source ~/.bashrc

# STEP 3A: SET HADOOP JAVA PATH
echo 'export JAVA_HOME=/usr/lib/jvm/java-11-openjdk-amd64' >> "$HADOOP_HOME/etc/hadoop/hadoop-env.sh"
hadoop version

# STEP 4: CONFIGURE PASSWORDLESS SSH
sudo systemctl enable --now ssh
mkdir -p ~/.ssh
chmod 700 ~/.ssh
[ -f ~/.ssh/id_rsa ] || ssh-keygen -t rsa -N "" -f ~/.ssh/id_rsa
cat ~/.ssh/id_rsa.pub >> ~/.ssh/authorized_keys
chmod 600 ~/.ssh/authorized_keys
ssh -o StrictHostKeyChecking=accept-new localhost
# Type exit after SSH connects
exit

# STEP 5: CONFIGURE core-site.xml
cat > "$HADOOP_HOME/etc/hadoop/core-site.xml" <<'EOF'
<configuration>
    <property>
        <name>fs.defaultFS</name>
        <value>hdfs://localhost:9000</value>
    </property>
</configuration>
EOF

# STEP 6: CONFIGURE hdfs-site.xml
mkdir -p "$HADOOP_HOME/data/namenode"
mkdir -p "$HADOOP_HOME/data/datanode"

cat > "$HADOOP_HOME/etc/hadoop/hdfs-site.xml" <<'EOF'
<configuration>
    <property>
        <name>dfs.replication</name>
        <value>1</value>
    </property>
    <property>
        <name>dfs.namenode.name.dir</name>
        <value>file:///usr/local/hadoop/data/namenode</value>
    </property>
    <property>
        <name>dfs.datanode.data.dir</name>
        <value>file:///usr/local/hadoop/data/datanode</value>
    </property>
</configuration>
EOF

# STEP 7: CONFIGURE MAPREDUCE
cat > "$HADOOP_HOME/etc/hadoop/mapred-site.xml" <<'EOF'
<configuration>
    <property>
        <name>mapreduce.framework.name</name>
        <value>yarn</value>
    </property>
</configuration>
EOF

# STEP 8: CONFIGURE YARN
cat > "$HADOOP_HOME/etc/hadoop/yarn-site.xml" <<'EOF'
<configuration>
    <property>
        <name>yarn.nodemanager.aux-services</name>
        <value>mapreduce_shuffle</value>
    </property>
</configuration>
EOF

# STEP 9: FORMAT THE NAMENODE
# Only for a fresh setup; do not reformat existing HDFS data.
hdfs namenode -format

# STEP 10: START HADOOP SERVICES
start-dfs.sh
start-yarn.sh
jps

# STEP 11: CREATE SAMPLE INPUT FILE
echo "hello hadoop hello spark" > ~/file.txt

# STEP 12: CREATE HDFS INPUT DIRECTORY
hdfs dfs -mkdir -p /input

# STEP 13: UPLOAD INPUT FILE
hdfs dfs -put -f ~/file.txt /input/

# STEP 14: RUN MAPREDUCE WORD COUNT
hdfs dfs -rm -r -f /output
hadoop jar "$HADOOP_HOME/share/hadoop/mapreduce/hadoop-mapreduce-examples-3.3.6.jar" wordcount /input /output

# STEP 15: DISPLAY RESULTS
hdfs dfs -cat /output/part-r-00000

# EXPECTED OUTPUT:
# hadoop  1
# hello   2
# spark   1
